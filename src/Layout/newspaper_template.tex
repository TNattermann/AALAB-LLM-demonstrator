% --- Custom Command for a 'Headline' Title ---
\newcommand{\newspaperheadline}[1]{
	\centering
	\fontsize{25pt}{25pt}\selectfont\textbf{#1}\par\vspace*{-15pt}
	\rule{\linewidth}{1.5pt} % Thick line under the headline
	\par\vskip 2pt
}

% --- Custom Command for the 'Dateline/Byline' ---
\newcommand{\byline}[1]{
	\centering
	\normalsize\textit{#1}\par\vspace*{-4pt}
	\rule{\linewidth}{0.4pt} % Thin line under the byline
	\par\vskip 10pt
}
